للوصول بالمنصة إلى مستوى **أنظمة المخططات الهندسية الاحترافية (مثل ZXW / Borneo Schematics / Wuxinji)** مدمجةً مع **غرفة العمليات الحية والذكاء الاصطناعي المحلي**، كان ينقصها 5 ركائز هندسية جوهرية تم بناؤها بالكامل في هذا التحديث (**الإصدار v13.0 — Enterprise BoardView, Schematics & Micro-Electronics Atlas**):

1. **قاعدة البيانات الهندسية الشاملة للهواتف (`hardware_atlas.py`)**: تغطي أشهر الموديلات العالمية من **Apple, Samsung, Xiaomi/Redmi/POCO, Realme/Oppo, Huawei/Honor, Infinix/Nokia** بتفاصيل المعالجات، آيسيهات الباور والشحن والصوت والشبكة، نقاط التيست بوينت (Testpoints)، وأوامر البرمجة والتفليش الحقيقية (`mtkclient`, `edl`, `Odin`, `libimobiledevice`, `checkm8`).
2. **محرك المخططات التفاعلية (`schematics_engine.py` + `templates/schematics.html`)**: توليد مخططات بوردة تفاعلية (Interactive SVG BoardView) ومخططات تسلسل الطاقة (Power Sequence Block Diagrams) قابلة للنقر على كل رجل (Pin) وكونكتر وآيسي لعرض اسم المسار (Net Name)، قيمة ممانعة الدايود المرجعية (mV)، الجهد الكهربائي (V)، والأعراض عند التلف.
3. **محلل سحب الباور سبلاي وتسلسل الإقلاع (`power_analyzer.py`)**: محرك تشخيص إلكتروني يحلل سحب الأمبير (mA) قبل الضغط على زر الباور (التسريب والقصر) وبعد الضغط عليه (مراحل إقلاع المعالج والرام والذاكرة) لكل معمارية.
4. **ماسح منافذ USB الحقيقي على جهاز الفني (`usb_scanner.py`)**: فحص المنافذ المتصلة فعلياً بنظام التشغيل (Windows/Linux) والتعرف التلقائي على وضعيات البوت والتفليش (`BROM`, `EDL 9008`, `ODIN`, `DFU`).
5. **بوابة متابعة العميل عبر QR (`templates/portal.html`)**: صفحة عامة مؤمّنة برمز التذكرة تتيح للعميل مسح رمز الـ QR من تقرير الـ PDF لمتابعة حالة جهازه وتكلفة الإصلاح لحظياً.

```python
def identify_usb_device(vid: str, pid: str) -> Dict:
    vid_clean = vid.strip().upper().replace("0X", "")
    pid_clean = pid.strip().upper().replace("0X", "")
    for sig in USB_HARDWARE_SIGNATURES:
        if sig["vid"] == vid_clean and sig["pid"] == pid_clean:
            return {"matched": True, **sig}
    return {
        "matched": False,
        "vid": vid_clean,
        "pid": pid_clean,
        "chipset": "Unknown / Generic USB",
        "mode": "Standard MTP / Composite",
        "description": "جهاز USB غير مدرج في وضعيات التفليش الهندسي الخاصة",
        "recommended_tools": ["Device Manager / lsusb", "ADB"],
        "risk_level": "low",
    }


def evaluate_diode_reading(rail_code: str, measured_v: float) -> Dict:
    rail = next((r for r in DIODE_REFERENCE_RAILS if r["rail_code"].lower() == rail_code.lower()), None)
    if not rail:
        if measured_v < 0.050:
            status = "SHORT_CIRCUIT"
            diagnosis = "قصر صريح (Short to GND) — يجب حقن جهد منخفض (1V / 2A) لتحديد المكون الساخن"
        elif measured_v > 1.500:
            status = "OPEN_LINE"
            diagnosis = "قطع في المسار (Open Line / OL) — افحص المقاومات والملفات المتصلة بالخط"
        else:
            status = "NORMAL_RANGE"
            diagnosis = "القراءة ضمن النطاق الممانعي المقبول مبدئياً"
        return {
            "rail_code": rail_code,
            "measured_v": measured_v,
            "status": status,
            "diagnosis": diagnosis,
        }

    if measured_v <= rail["short_threshold"]:
        status = "SHORT_CIRCUIT"
        diagnosis = f"قصر حرج على {rail['name_ar']}! المشتبه بهم: {rail['short_suspects']}"
    elif measured_v < rail["min_v"]:
        status = "LEAKAGE_LOW"
        diagnosis = f"تسريب (ممانعة منخفضة عن الحد الأدنى {rail['min_v']}V). افحص: {rail['short_suspects']}"
    elif measured_v > rail["max_v"] or measured_v >= 1.500:
        status = "OPEN_LINE"
        diagnosis = f"قطع أو ممانعة مرتفعة على {rail['name_ar']}. السبب المحتمل: {rail['open_suspects']}"
    else:
        status = "HEALTHY"
        diagnosis = f"مسار سليم 100% — القيمة ({measured_v}V) تقع ضمن النطاق المرجعي ({rail['min_v']}V - {rail['max_v']}V)"

    return {
        "rail_code": rail["rail_code"],
        "rail_name_ar": rail["name_ar"],
        "measured_v": measured_v,
        "expected_range": f"{rail['min_v']}V – {rail['max_v']}V",
        "status": status,
        "diagnosis": diagnosis,
    }


def search_device_atlas(query: str = "", brand: str = "") -> List[Dict]:
    q = query.strip().lower()
    b = brand.strip().lower()
    results = []
    for dev in DEVICE_MODELS_ATLAS:
        if b and b not in dev["brand"].lower():
            continue
        if q:
            haystack = (
                f"{dev['brand']} {dev['model']} {dev['soc']} "
                + " ".join(dev["ics"].values())
            ).lower()
            if q not in haystack:
                continue
        results.append(dev)
    return results
```

```python
"""DC Power Supply Boot-Sequence & Current Draw Diagnostic Engine.
Analyzes pre-boot leakage (before pressing power key) and post-boot current draw (mA)
across Apple, Qualcomm, and MediaTek architectures.

def analyze_dc_power_draw(

    pre_boot_ma: float,
    post_boot_ma: float,
    boot_behavior: str = "stuck",
) -> Dict:
    """
    architecture: 'apple' | 'qualcomm' | 'mediatek'
    pre_boot_ma: current draw in mA as soon as DC power supply is connected (before pressing power)
    post_boot_ma: current draw in mA after pressing the power button
    boot_behavior: 'zero' | 'drops_to_zero' | 'stuck' | 'cycling'
    """
    arch = architecture.lower().strip()

    # 1. Pre-Boot Leakage / Primary Short Analysis (Before pressing Power Button)
    if pre_boot_ma >= 500:
        return {
            "stage": "PRE_BOOT_DEAD_SHORT",
            "severity": "critical",
            "fault_title": f"قصر صريح على خط التغذية الرئيسي (سحب {pre_boot_ma}mA قبل الضغط على الباور)",
            "suspect_rails": ["PP_VDD_MAIN / VPH_PWR", "PP_BATT_VCC / VBAT"],
```

### 5️⃣ ماسح منافذ USB/Serial الحقيقي للمختبر: `usb_scanner.py`

```python
"""Real Bench USB & Serial COM Port Scanner.
Uses pyserial (cross-platform) to detect live hardware devices plugged into the lab PC
and matches them against real smartphone bootloader VID:PID signatures.
"""
from typing import Dict, List
from hardware_atlas import identify_usb_device


def scan_connected_usb_ports() -> List[Dict]:
    """Scan active USB/Serial ports on the host machine without raising errors."""
    detected = []
    try:
        import serial.tools.list_ports
        ports = serial.tools.list_ports.comports()
        for p in ports:
            vid_hex = f"{p.vid:04X}" if p.vid is not None else ""
            pid_hex = f"{p.pid:04X}" if p.pid is not None else ""
            ident = identify_usb_device(vid_hex, pid_hex) if vid_hex and pid_hex else {"matched": False}
            detected.append({
                "port": p.device,
                "description": p.description or "USB Serial Device",
                "hwid": p.hwid or "",
                "vid": vid_hex,
                "pid": pid_hex,
                "manufacturer": p.manufacturer or ident.get("chipset", "Unknown"),
                "product": p.product or ident.get("mode", "Standard Device"),
                "identification": ident,
            })
    except Exception:
        pass
    return detected
```

```python
"""Multimodal Hardware Diagnostic Co-Pilot (Diode + DC Power Supply + USB + OpenCV + Local Ollama)."""
import base64
import json
import os
from typing import Dict, List, Optional
import httpx

OLLAMA_URL = os.getenv("OLLAMA_URL", "http://127.0.0.1:11434")
OLLAMA_VISION_MODEL = os.getenv("OLLAMA_VISION_MODEL", "llava")
OLLAMA_TEXT_MODEL = os.getenv("OLLAMA_TEXT_MODEL", "qwen2.5-coder:7b")


async def query_ollama_diagnostic(prompt: str, image_path: Optional[str] = None) -> Optional[str]:
    try:
        payload: Dict = {
            "model": OLLAMA_VISION_MODEL if image_path else OLLAMA_TEXT_MODEL,
            "prompt": prompt,
            "stream": False,
        }
        if image_path and os.path.exists(image_path):
            with open(image_path, "rb") as f:
                payload["images"] = [base64.b64encode(f.read()).decode("utf-8")]

        async with httpx.AsyncClient(timeout=8.0) as client:
            resp = await client.post(f"{OLLAMA_URL}/api/generate", json=payload)
            if resp.status_code == 200:
                return resp.json().get("response", "").strip()
    except Exception:
        return None
    return None


async def run_multimodal_diagnosis(
    case: dict,
    diode_readings: List[dict],
    usb_sessions: List[dict],
    vision_analyses: List[dict],
    power_logs: Optional[List[dict]] = None,
) -> Dict:
    evidence_summary = []
    actions = []
    confidence = 75
    primary_fault = f"فحص أولي للجهاز {case.get('device_model', '')}: {case.get('reported_issue', 'عطل عام')}"

    # 1. Diode Readings
    shorts = [d for d in diode_readings if d.get("status") == "SHORT_CIRCUIT"]
    opens = [d for d in diode_readings if d.get("status") == "OPEN_LINE"]

    if shorts:
        s = shorts[0]
        primary_fault = f"قصر صريح (Short Circuit) على مسار {s['rail_code']} بقيمة {s['measured_v']}V"
        confidence = 95
        evidence_summary.append({"source": "diode_mode", "detail": s["diagnosis"]})
        actions.extend([
            f"حقن جهد منخفض (1.0V - 1.5V / 2A) على مسار {s['rail_code']} لتحديد المكون الساخن",
            "استخدام بخاخ التبريد (Freeze Spray) أو الكاميرا الحرارية لعزل المكثف أو الآيسي المتسبب بالقصر",
        ])
    elif opens:
        o = opens[0]
        primary_fault = f"قطع في المسار (Open Line) على خط {o['rail_code']}"
        confidence = 90
        evidence_summary.append({"source": "diode_mode", "detail": o["diagnosis"]})
        actions.append(f"فحص المقاومات والملفات المتصلة على التوالي مع مسار {o['rail_code']} تحت الميكروسكوب")

    # 2. DC Power Supply Logs
    if power_logs:
        last_pwr = power_logs[0]
        evidence_summary.append({
            "source": "dc_power_supply",
            "detail": f"سحب قبل الباور {last_pwr['pre_boot_ma']}mA / بعد الباور {last_pwr['post_boot_ma']}mA — {last_pwr['fault_title']}",
        })
        if not shorts and last_pwr["stage"] != "BOOT_OS_STAGE":
            primary_fault = last_pwr["fault_title"]
            confidence = max(confidence, 92)

    # 3. USB Signatures
    if usb_sessions:
        last_usb = usb_sessions[0]
        mode = last_usb.get("mode", "")
        evidence_summary.append({"source": "usb_port", "detail": f"تم رصد وضعية {mode} ({last_usb.get('vid')}:{last_usb.get('pid')})"})
        if not shorts and ("BROM" in mode or "EDL" in mode):
            primary_fault = f"الجهاز عالق في وضع الطوارئ ({mode}) نتيجة تلف ملفات الإقلاع أو الذاكرة الداخلية"
            confidence = 92
            actions.extend([
                "سحب نسخة احتياطية من قطاعات الأمان (NVDATA / NVRAM / EFS) قبل أي عملية تفليش",
                "تمرير ملفات البوت والنظام المناسبة عبر واجهة التفليش الهندسية",
            ])

    # 4. OpenCV Vision Findings
    latest_image = None
    if vision_analyses:
        last_vis = vision_analyses[0]
        latest_image = last_vis.get("image_path")
        v_findings = json.loads(last_vis.get("findings_json") or "[]")
        for vf in v_findings:
            evidence_summary.append({"source": "opencv_vision", "detail": f"{vf.get('label_ar', vf['type'])} (العدد: {vf['count']})"})
            if vf["type"] == "corrosion":
                actions.append("تنظيف البوردة بمحلول التنظيف بالموجات فوق الصوتية وإعادة لحام الأرجل المتأكسدة بالفلكس")
            elif vf["type"] == "burn_marks":
                actions.append("رفع المكون المتفحم وفحص سلامة وسادات اللحام (Pads) أسفله قبل التركيب")

    if not actions:
        actions.append("إجراء فحص سحب التيار على الباور سبلاي وقياس ممانعات كونكتر البطارية والشحن")

    # 5. Optional Local Ollama Enrichment
    ollama_prompt = (
        f"أنت مهندس صيانة هواتف خبير. الجهاز: {case.get('device_model')}. "
        f"الشكوى: {case.get('reported_issue')}. "
        f"التشخيص المبدئي: {primary_fault}. "
        f"أعطِ نصيحة هندسية مركزة في سطرين باللغة العربية."
    )
    ollama_advice = await query_ollama_diagnostic(ollama_prompt, latest_image)
    engine_used = "GRE-Multimodal + Ollama Local AI" if ollama_advice else "GRE-Multimodal Hardware Engine v13"
    if ollama_advice:
        actions.append(f"توصية الذكاء الاصطناعي المحلي: {ollama_advice}")

    return {
        "final_diagnosis": primary_fault,
        "confidence_pct": confidence,
        "evidence": evidence_summary,
        "recommended_actions": actions,
        "ai_engine": engine_used,
    }
```

```python
@app.websocket("/ws/live/{token}")
async def ws_live_vision(websocket: WebSocket, token: str):
    token_jwt = websocket.query_params.get("token")
    if not token_jwt:
        await websocket.close(code=4001, reason="missing auth")
        return
    try:
        decode_token_raw(token_jwt)
    except Exception:
        await websocket.close(code=4001, reason="invalid token")
        return

    session = live_vision.get_session(token)
    if not session:
        await websocket.close(code=4004, reason="session expired")
        return

    await websocket.accept()
    frame_counter = 0
    real_session_id = session["session_id"]

    try:
        while True:
            raw = await websocket.receive_text()
            try:
                msg = json.loads(raw)
            except Exception:
                continue
            if msg.get("type") != "frame":
                continue

            frame_counter += 1
            frame = live_vision.decode_base64_frame(msg.get("image", ""))
            if frame is None:
                await websocket.send_json({"type": "error", "message": "bad frame"})
                continue

            if frame_counter % ANALYZE_EVERY_N_FRAMES != 0:
                live_vision.increment_frame(token, 0)
                continue

            result = live_vision.analyze_frame(frame, full_analysis=(frame_counter % 30 == 0))
            live_vision.increment_frame(token, result["anomalies"])

            saved_path = None
            is_flagged = 0
            if result["anomalies"] > 0:
                saved_path = live_vision.save_flagged_frame(frame)
                is_flagged = 1

            async with aiosqlite.connect(DB_FILE) as db:
                await db.execute(
                    """INSERT INTO live_frames
                       (session_id, frame_number, findings_json, anomalies_count, saved_image_path, is_flagged)
                       VALUES (?, ?, ?, ?, ?, ?)""",
                    (
                        real_session_id, frame_counter,
                        json.dumps(result["findings"], ensure_ascii=False),
                        result["anomalies"], saved_path, is_flagged,
                    ),
                )
                await db.commit()

            await websocket.send_json({
                "type": "analysis",
                "frame": frame_counter,
                "findings": result["findings"],
                "anomalies": result["anomalies"],
                "analysis_ms": result["analysis_ms"],
                "flagged": is_flagged == 1,
            })
    except WebSocketDisconnect:
        pass
    finally:
        sess = live_vision.unregister_session(token)
        if sess:
            async with aiosqlite.connect(DB_FILE) as db:
                await db.execute(
                    """UPDATE live_sessions SET ended_at = CURRENT_TIMESTAMP,
                       frame_count = ?, total_anomalies = ?, status = 'ended'
                       WHERE session_token = ?""",
                    (sess["frame_count"], sess["anomalies_total"], token),
                )
                await db.commit()

    await websocket.accept()
    try:
        history = collab_manager.get_history(case_id, limit=100)
        await websocket.send_json({
            "type": "history",
            "messages": history,
            "participants": collab_manager.get_participants(case_id),
        })
        await collab_manager.join(case_id, user, websocket)

        while True:
            raw = await websocket.receive_text()
            try:
                msg = json.loads(raw)
            except Exception:
                continue
            mtype = msg.get("type")
            if mtype == "chat":
                await collab_manager.chat(case_id, user, msg.get("text", ""))
            elif mtype == "cursor":
                await collab_manager.cursor(case_id, user, float(msg.get("x", 0)), float(msg.get("y", 0)))
            elif mtype == "annotation":
                await collab_manager.annotate(case_id, user, msg.get("annotation", {}))
            elif mtype == "ping":
                await websocket.send_json({"type": "pong"})
    except WebSocketDisconnect:
        pass
    finally:
        await collab_manager.leave(case_id, user, websocket)
```
